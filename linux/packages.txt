# 0x58 — apt base packages for a headless Linux box (Debian trixie / Ubuntu)
#
# Scope: apt owns ONLY the base system. Every language runtime and dev CLI comes
# from mise (dotfiles/mise/config.toml), which is already cross-platform. Do not
# duplicate anything mise manages — node, go, python, kubectl, terraform, gh,
# ripgrep, fd, bat, fzf, jq, yq, shellcheck, ... Listing them here just
# shadows mise's shims with older Debian builds and makes upgrades ambiguous.
#
# Format: one package per line. '#' comments and blank lines are ignored.
# Consumed by macos/restore.sh -> install_packages().
# Inventory the live system with:  apt-mark showmanual

openssh-server

# === Agent sandboxing — REQUIRED, not optional ===
# Claude Code's Linux sandbox is bubblewrap + socat, and Codex's is bubblewrap.
# linux/claude-settings.json sets failIfUnavailable=true, which means a missing
# bwrap is a hard startup failure rather than a silent fall back to running
# model-generated commands unsandboxed. Removing these does not soften the
# posture — it stops Claude Code from running at all. That is the intent.
bubblewrap
socat

# === Server observability ===
# dotfiles/mise/config.toml notes these two have no macOS aqua binaries and are
# "kept in brew on Mac" — but no mise entry exists for them, so on Linux apt is
# the one thing that actually provides them.
btop
ncdu

# === Containers — for your shell, CI, AND agents ===
# Rootless podman provides the Docker-compatible API socket testcontainers-go looks
# for. Until 2026-08-04 it was deliberately unreachable from inside an agent
# sandbox; `allowAllUnixSockets` is now on, so agents run the container-backed
# tiers themselves. That switch is all-or-nothing on Linux — seccomp cannot filter
# a socket by path, and `allowUnixSockets` says "macOS only ... ignored on Linux"
# in its own schema.
#
# What it costs is NOT mainly the SSH agent, which is what this comment used to
# warn about: signing and pushing already use an on-disk key, so no Touch ID tap
# was being protected. The real cost is that the runtime executes OUTSIDE the
# sandbox, so image pulls ignore the egress allowlist and bind mounts reach denied
# paths. Read linux/sandbox.md before changing this.
podman
uidmap         # newuidmap/newgidmap — rootless podman cannot map subuids without it
passt          # pasta: podman 5's default rootless network backend
podman-remote  # `podman --remote`; also what a toolbox container needs to drive
               # the host socket when loopback TCP is blocked in-session

# === Agent-driven cluster/database work ===
# Installed by hand inside a throwaway container the first time an agent needed
# them, which is a bad place for them to live — a fresh container starts from
# nothing and the apt install is repeated every run.
procps            # ps/free; Priority:important so already present, listed to pin it
postgresql-client # psql against a service in kind or a compose stack
# NOTE: jq is deliberately NOT here. mise owns it (see the header) and an apt jq
# would shadow the shim with an older build.
